# AS-2-2-2


## #Grundkonfig

conf t
no ip domain-lookup

hostname AS-2-2-2

username cisco priv 15 secret cisco

ip domain-name 5CN.at

crypto key gen rsa mod 1024
ip ssh version 2

line vty 0 15
exec-timeout 0 0
loggin sync 
login local 
transport input telnet ssh
exit

line con 0 
exec-timeout 0 0
loggin sync 
login local 
exit

## #Interface konfig

int g0/1
no shut
ip address 10.0.79.9 255.255.255.0
exit
int g0/2
no shut
ip address 10.0.89.9 255.255.255.0

int lo1
no shut
ip address 9.9.9.9 255.255.255.255
exit


## #OSPF
router ospf 1
router-id 9.9.9.9

network 9.9.9.9 0.0.0.0 area 1

network 10.0.79.9 0.0.0.3 area 1
network 10.0.89.9 0.0.0.3 area 1

exit
