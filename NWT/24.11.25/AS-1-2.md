# AS-1-2


## #Grundkonfig

conf t
no ip domain-lookup

hostname AS-1-2

username cisco priv 15 secret cisco

ip domain-name 5CN.at

crypto key gen rsa mod 1024
ip ssh version 2

line vty 0 15
exec-timeout 0 0
loggin sync 
login local 
transport input telnet ssh
exit

line con 0 
exec-timeout 0 0
loggin sync 
login local 
exit

int g 0/0

int lo1
