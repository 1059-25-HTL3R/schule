hostname AS-1-5

username cisco priv 15 secret cisco

ip domain-name 5CN.at

line vty 0 15
exec-timeout 0 0
loggin sync 
login local 
transport input telnet ssh
exit

line con 0 
exec-timeout 0 0
loggin sync 
login local 
exit
